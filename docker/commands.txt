# Build Docker image
docker build -t lab2 .

# Run Docker container
docker run -d -p 8080:80 --name tr1 lab2

# Check running containers
docker ps

# Get image ID
docker images

# Tag image for Docker Hub
docker tag lab2 username/imagename:latest

# Login to Docker Hub
docker login

# Push image to Docker Hub
docker push username/imagename:latest

# Pull image from Docker Hub
docker pull username/imagename:latest

# Stop container
docker stop tr1

# Remove container
docker rm tr1
