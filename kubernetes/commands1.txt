# =================================
# 1. KUBERNETES CLUSTER
# =================================

# Check Kubernetes version
kubectl version

# Check available nodes
kubectl get nodes

# Check cluster information
kubectl cluster-info


# =================================
# 2. DOCKER COMMANDS
# =================================

# Build Docker image
docker build -t myapp .

# Run Docker container
docker run -d --name mycontainer -p 8080:80 myapp

# List Docker images
docker images

# List running containers
docker ps


# =================================
# 3. DEPLOYMENT COMMANDS
# =================================

# Create deployment
kubectl apply -f deployment.yaml

# Check deployments
kubectl get deployments

# Check pods
kubectl get pods

# Display detailed pod information
kubectl get pods -o wide


# =================================
# 4. SERVICE COMMANDS
# =================================

# Create service
kubectl apply -f service.yaml

# Check services
kubectl get services

# Check all endpoints
kubectl get endpoints

# Check specific service endpoints
kubectl get endpoints myapp-service


# =================================
# 5. ACCESS APPLICATION
# =================================

# Port forwarding
kubectl port-forward service/myapp-service 8080:80

# Browser URL:
# http://localhost:8080

# NodePort URL:
# http://localhost:30080


# =================================
# 6. SCALING DEPLOYMENT
# =================================

# Increase replicas to 7
kubectl scale deployment myapp-deployment --replicas=7

# Verify pods
kubectl get pods


# =================================
# 7. DELETE POD
# =================================

# Delete a specific pod
kubectl delete pod POD_NAME

# Verify automatic pod recreation
kubectl get pods


# =================================
# 8. VERIFY SERVICE AGAIN
# =================================

kubectl get services
kubectl get endpoints myapp-service
kubectl get pods -o wide
